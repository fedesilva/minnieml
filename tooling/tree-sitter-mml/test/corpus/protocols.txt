protocol Render 'T =
  fn render(label: String, value: 'T): Unit;
;

instance Render Int =
  fn render(label, value) =
    print label;
    println (int_to_str value);
  ;
;

instance Render Bool =
  fn render(label, value) =
    print label;
    if value then println "true";
    else println "false";
    ;
  ;
;

pub fn main(): Int = render "answer: " 42; 0;;
