// Proposed standard-library syntax, including future polymorphic features.
// Protocols: ../language/protocols.md
// Unions: ../language/unions.md; enums: ../language/enums.md
// Match syntax is illustrative; branches and expressions use semicolon terminators.

// Represents an optional value
// You can have maybe one value or none.
union Maybe 'T =
  | One 'T
  | None
;

// Puts Maybe members in the current scope to allow using them unqualified
// (One instead of Maybe.One)
use Maybe.*;

// Function type variables are introduced by their signatures.
fn empty(x: Maybe 'T): Bool =
  match x with
    | None -> true;
    | _    -> false;
  ;
;

fn non_empty(maybe: Maybe 'T): Bool =
  not (empty maybe);
;

// Fixing the error type leaves Result 'E with one parameter for a future Functor instance.
union Result 'E 'T =
  | Ok 'T
  | Err 'E
;

// Enums contain integer-backed members without payloads.
enum Ordering =
  | Less
  | Equal
  | Greater
;

let equal_ordering = Ordering.Equal;

// Semigroup represents an associative binary operation.
protocol Semigroup 'T =
  fn combine(left: 'T, right: 'T): 'T;
;

// Monoid requires 'T to also satisfy Semigroup.
protocol Monoid 'T
  where 'T: Semigroup =
  fn empty(): 'T;
;

// Functor represents a structure that can be mapped over.
protocol Functor 'F =
  fn map(f: 'A -> 'B, values: 'F 'A): 'F 'B;
;

// Apply requires 'F to also satisfy Functor.
protocol Apply 'F
  where 'F: Functor =
  fn ap(functions: 'F ('A -> 'B), values: 'F 'A): 'F 'B;
;

// Applicative requires 'F to also satisfy Apply.
protocol Applicative 'F
  where 'F: Apply =
  fn pure(value: 'A): 'F 'A;
;

// FlatMap requires 'F to also satisfy Functor.
protocol FlatMap 'F
  where 'F: Functor =
  fn flat_map(f: 'A -> 'F 'B, values: 'F 'A): 'F 'B;
;

// Monad requires 'M to also satisfy Functor, FlatMap, and Applicative.
protocol Monad 'M
  where 'M: Functor & FlatMap & Applicative;

protocol Foldable 'F =
  fn fold_left(f: 'B -> 'A -> 'B, initial: 'B, values: 'F 'A): 'B;
  fn fold_right(f: 'A -> 'B -> 'B, initial: 'B, values: 'F 'A): 'B;
;

// Illustrative additive instance, not a settled standard-library choice for bare Int.
// Canonical instances allow one Semigroup Int; supporting both addition and multiplication
// requires distinct nominal types, such as Sum and Product, with their own instances.
instance Semigroup Int =
  fn combine(left: Int, right: Int): Int = left + right;;
;

// The matching illustrative Monoid uses 0 as the additive identity.
instance Monoid Int =
  fn empty(): Int = 0;;
;

// Maybe implements the Functor protocol
instance Functor Maybe =
  fn map(f: 'A -> 'B, maybe: Maybe 'A): Maybe 'B =
    match maybe with
      | One x -> One (f x);
      | None  -> None;
    ;
  ;
;

// Maybe implements the Apply protocol
instance Apply Maybe =
  fn ap(f: Maybe ('A -> 'B), maybe: Maybe 'A): Maybe 'B =
    match f with
      | One g -> map g maybe;
      | None  -> None;
    ;
  ;
;

// Maybe implements the FlatMap protocol
instance FlatMap Maybe =
  fn flat_map(f: 'A -> Maybe 'B, maybe: Maybe 'A): Maybe 'B =
    match maybe with
      | One x -> f x;
      | None  -> None;
    ;
  ;
;

// Maybe implements the Applicative protocol
instance Applicative Maybe =
  fn pure(x: 'A): Maybe 'A = One x;;
;

// Monad's requirements are supplied by the separate instances above.
instance Monad Maybe;

protocol Something 'T
  where 'T: Semigroup =
  fn combine_with_plus(left: 'T, right: 'T): 'T = left + right;;
  // Every visible declaration of a symbol and arity must agree on grouping.
  // These match Num's binary + and the standard integer >> operator.
  op +(left: 'T, right: 'T): 'T 60 left;
  op >>(left: 'T, right: 'T): 'T 55 left;
;

// Future parameterized instance using the proposed union syntax.
union Option 'T =
  | Some 'T
  | None
;

instance Semigroup (Option 'T)
  where 'T: Semigroup =
  fn combine(left: Option 'T, right: Option 'T): Option 'T =
    match left with
      | Option.None -> right;
      | Option.Some x ->
        match right with
          | Option.None -> left;
          | Option.Some y -> Option.Some (combine x y);
        ;
    ;
  ;
;
