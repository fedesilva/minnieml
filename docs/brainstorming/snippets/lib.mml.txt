// Proposed standard-library syntax, including future polymorphic features.
// Protocols: ../language/protocols.md
// Unions: ../language/unions.md; enums: ../language/enums.md
// Match syntax is illustrative; branches and expressions use semicolon terminators.

// Represents an optional value
// You can have maybe one value or none.
union Maybe 'T =
  | One 'T
  | None
;

// Puts Maybe members in the current scope to allow using them unqualified
// (One instead of Maybe.One)
use Maybe.*;

// Function type variables are introduced by their signatures.
fn empty(x: Maybe 'T): Bool =
  match x with
    | None -> true;
    | _    -> false;
  ;
;

fn nonEmpty(maybe: Maybe 'T): Bool =
  not (empty maybe);
;

union Result 'T 'E =
  | Ok 'T
  | Err 'E
;

// Enums contain integer-backed members without payloads.
enum Ordering =
  | Less
  | Equal
  | Greater
;

let equalOrdering = Ordering.Equal;

// Semigroup represents an associative binary operation.
protocol Semigroup 'T =
  fn combine(left: 'T, right: 'T): 'T;
;

// Monoid requires 'T to also satisfy Semigroup.
protocol Monoid 'T: Semigroup =
  fn empty(): 'T;
;

// Functor represents a structure that can be mapped over.
protocol Functor 'F =
  fn map(f: 'A -> 'B, values: 'F 'A): 'F 'B;
;

// Apply requires 'F to also satisfy Functor.
protocol Apply 'F: Functor =
  fn ap(functions: 'F ('A -> 'B), values: 'F 'A): 'F 'B;
;

// Applicative requires 'F to also satisfy Apply.
protocol Applicative 'F: Apply =
  fn pure(value: 'A): 'F 'A;
;

// FlatMap requires 'F to also satisfy Functor.
protocol FlatMap 'F: Functor =
  fn flatMap(f: 'A -> 'F 'B, values: 'F 'A): 'F 'B;
;

// Monad requires 'M to also satisfy Functor, FlatMap, and Applicative.
protocol Monad 'M: Functor & FlatMap & Applicative;

protocol Foldable 'F =
  fn foldLeft(f: 'B -> 'A -> 'B, initial: 'B, values: 'F 'A): 'B;
  fn foldRight(f: 'A -> 'B -> 'B, initial: 'B, values: 'F 'A): 'B;
;

// Int forms a semigroup under addition
instance Semigroup Int =
  fn combine(left: Int, right: Int): Int = left + right;;
;

// Int forms a monoid under addition with 0 as identity
instance Monoid Int =
  fn empty(): Int = 0;;
;

// Maybe implements the Functor protocol
instance Functor Maybe =
  fn map(f: 'A -> 'B, maybe: Maybe 'A): Maybe 'B =
    match maybe with
      | One x -> One (f x);
      | None  -> None;
    ;
  ;
;

// Maybe implements the Apply protocol
instance Apply Maybe =
  fn ap(f: Maybe ('A -> 'B), maybe: Maybe 'A): Maybe 'B =
    match f with
      | One g -> map g maybe;
      | None  -> None;
    ;
  ;
;

// Maybe implements the FlatMap protocol
instance FlatMap Maybe =
  fn flatMap(f: 'A -> Maybe 'B, maybe: Maybe 'A): Maybe 'B =
    match maybe with
      | One x -> f x;
      | None  -> None;
    ;
  ;
;

// Maybe implements the Applicative protocol
instance Applicative Maybe =
  fn pure(x: 'A): Maybe 'A = One x;;
;

// Maybe implements the Monad protocol
instance Monad Maybe =
  fn flatMap(f: 'A -> Maybe 'B, maybe: Maybe 'A): Maybe 'B =
    match maybe with
      | One x -> f x;
      | None  -> None;
    ;
  ;

  fn pure(x: 'A): Maybe 'A = One x;;
;

protocol Something 'T: Semigroup =
  fn combine(left: 'T, right: 'T): 'T = left + right;;
  // Example of operator definitions using protocols
  // Here the precedence and associativity are randomly
  // selected for demonstration only.
  op +(left: 'T, right: 'T): 'T 20 right;
  op >>(left: 'T, right: 'T): 'T 25 left;
;

// Future parameterized instance using the proposed union syntax.
union Option 'T =
  | Some 'T
  | None
;

instance Semigroup (Option 'T: Semigroup) =
  fn combine(left: Option 'T, right: Option 'T): Option 'T =
    match left with
      | Option.None -> right;
      | Option.Some x ->
        match right with
          | Option.None -> left;
          | Option.Some y -> Option.Some (combine x y);
        ;
    ;
  ;
;
